async function loadAll(){
  if(!state.session){render();return}
  const qs=await Promise.all([
    sb.from('partners').select('*').order('display_name'),
    sb.from('attendance').select('*').order('work_date',{ascending:false}).limit(1500),
    sb.from('documents').select('*').order('expires_on',{ascending:true,nullsFirst:false}),
    sb.from('worker_profiles').select('*'),
    sb.from('profile_photos').select('*')
  ]);
  const err=qs.find(x=>x.error)?.error;if(err){toast('同期エラー: '+err.message);return}
  [state.partners,state.attendance,state.documents,state.workers,state.photos]=qs.map(x=>x.data||[]);
  syncPill();render();
}
